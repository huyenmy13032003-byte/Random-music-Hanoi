// Chế độ Admin riêng cho máy của bạn const VIP_KEY = "chapter1_hanoi_vip"; const params = new URLSearchParams(window.location.search); if (params.get("vip") === "1") { localStorage.setItem(VIP_KEY, "true"); } else if (params.get("vip") === "0") { localStorage.removeItem(VIP_KEY); localStorage.removeItem(STORAGE_KEY); } const isVip = localStorage.getItem(VIP_KEY) === "true"; // Kiểm tra tổng số lần đã mở/quét mã trên thiết bị này let usedPlays = parseInt(localStorage.getItem(STORAGE_KEY) || "0", 10); if (!isVip && usedPlays >= MAX_PLAYS) { // Đã dùng hết 2 lượt: khóa hoàn toàn trình phát, chỉ hiện thông báo ytControls.style.display = "none"; tiktokBox.style.display = "none"; rerollArea.style.display = "none"; statusEl.textContent = "Bạn đã hết số lần trải nghiệm"; coffeeNotice.style.display = "block"; } else { if (isVip) { rerollInfo.textContent = "Chế độ Admin • Không giới hạn lượt"; nextBtn.addEventListener("click", () => location.reload()); } else { // Còn lượt: ghi nhận thêm 1 lượt ngay khi mở trang usedPlays += 1; localStorage.setItem(STORAGE_KEY, String(usedPlays)); if (usedPlays >= MAX_PLAYS) { // Đây là lượt cuối cùng (lượt 2/2) rerollArea.style.display = "none"; coffeeNotice.style.display = "block"; } else { const conLai = MAX_PLAYS - usedPlays; rerollInfo.textContent = `Lượt ${usedPlays}/${MAX_PLAYS} • Còn ${conLai} lượt đổi bài`; nextBtn.addEventListener("click", () => location.reload()); } }
